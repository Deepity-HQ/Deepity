112 // Registry: linear vector, not a hash map -- small layer counts, and
113 // preserves insertion order (== network layer order) for printing.
114 inline std::vector<std::pair<const void *, ProfileEntry>> &ProfileRegistry()
115 {
116 static std::vector<std::pair<const void *, ProfileEntry>> registry;
117 return registry;
118 }
119
120 // Looks up (or creates, on first call) this layer's accumulator. Label
121 // is generated from the layer's own GetInputSize()/GetOutputSize() --
122 // works for any layer type with those methods (DiscriminativePCLayer,
123 // ConvPCLayer, ...) via duck-typed template, no coupling to a specific
124 // class or forward declaration needed.
125 template <typename LayerT>
126 inline ProfileAccumulator &LayerProfile(LayerT *layer)
127 {
128 auto &registry = ProfileRegistry();
129 for (auto &entry : registry)
130 if (entry.first == layer)
131 return entry.second.acc;
132
133 ProfileEntry entry;
134 entry.label = "layer(" + std::to_string(layer->GetInputSize()) + "->"
135 + std::to_string(layer->GetOutputSize()) + ") @ " + std::to_string((size_t)layer);
136 registry.emplace_back(layer, std::move(entry));
137 return registry.back().second.acc;
138 }

140 // Call once, after the training run, with an INDEPENDENTLY measured
141 // wall-clock duration (std::chrono around the whole run in the harness,
142 // NOT derived from these accumulators) -- this is the reconciliation
143 // check from review point 2.
144 inline void PrintAllProfiles(double wallClockSeconds)
145 {
146 double grandTotal = 0.0;
147 for (auto &entry : ProfileRegistry())
148 {
149 entry.second.acc.Print(entry.second.label);

151 }
152 std::cout << "\n=== Reconciliation ===\n";
153 std::cout << " Wall-clock (measured around whole run): " << std::fixed << std::setprecision(4)
154 << wallClockSeconds << " s\n";
155 std::cout << " Sum of all layers' phase sums: " << grandTotal << " s\n";
156 std::cout << " Unaccounted-for time: " << (wallClockSeconds - grandTotal) << " s"
157 << " (" << (wallClockSeconds > 0 ? 100.0 * (wallClockSeconds - grandTotal) / wallClockSeconds : 0.0)
