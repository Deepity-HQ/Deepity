97 inline void *HugePageAllocPortable(size_t size, bool &succeeded)
98 {
99 succeeded = false;
100#if defined(_WIN32)
101 void *ptr = VirtualAlloc(nullptr, size, MEM_RESERVE | MEM_COMMIT | MEM_LARGE_PAGES, PAGE_READWRITE);
102 if (ptr)
103 succeeded = true;
104 return ptr;
105#else
106 void *ptr = mmap(nullptr, size, PROT_READ | PROT_WRITE,
107 MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
108 if (ptr == MAP_FAILED)
109 return nullptr;
110
111 // Best-effort hint -- if this fails, we still have valid,
112 // page-aligned (though not huge-page-backed) memory from the
113 // mmap itself, so this isn't treated as a hard failure.
114 madvise(ptr, size, MADV_HUGEPAGE);
115 succeeded = true;
116 return ptr;
117#endif
118 }

175 explicit MemoryArena(size_t total_floats, bool use_huge_pages = false)
176 : used_huge_pages(false)
177 {
178 // Guarantee total capacity is a multiple of 64 bytes
179 capacity_bytes = (total_floats * sizeof(float) + 63) & ~63;
180
181 if (use_huge_pages)
182 {
183 bool succeeded = false;
184 base_ptr = static_cast<float *>(detail::HugePageAllocPortable(capacity_bytes, succeeded));
185 used_huge_pages = succeeded;
186 }
187
188 if (!used_huge_pages)
189 {
190 // Either huge pages weren't requested, or the request
191 // wasn't honored (mmap/VirtualAlloc failed, or the
192 // system lacks the required configuration/privilege) --
193 // fall back to the standard aligned allocator rather
194 // than treating this as a hard failure.
195 base_ptr = static_cast<float *>(detail::AlignedAllocPortable(64, capacity_bytes));
196 }
197
198 if (!base_ptr)
199 {
200 throw std::bad_alloc();
201 }
202 offset_bytes = 0;
203 }
204
208 ~MemoryArena()
209 {
210 if (base_ptr)
211 {
212 if (used_huge_pages)
213 detail::HugePageFreePortable(base_ptr, capacity_bytes);
214 else
215 detail::AlignedFreePortable(base_ptr);
216 }
217 }
