63 double PhaseSum() const noexcept
64 {
65 // Sub-phases only -- does NOT include calculateStateTotal/
66 // updateStateTotal, since those measure the SAME work the
67 // sub-phases already measure (they'd double-count if summed
68 // together). calculateStateTotal/updateStateTotal exist purely
69 // as an independent top-level cross-check against the sub-phase
70 // sum, and against everything outside CalculateState/UpdateState
71 // (e.g. UpdateWeights, UpdatePrecision, caller-side overhead).
72 return errorConstruction + energyCalculation + forwardGemm + biasAdd + activationForward
73 + activationDerivative + localErrorTerm + bottomUpConstruction + feedbackGemm + saxpyUpdate
74 + updateWeightsTotal + updatePrecisionTotal;
75 }

125 // Registry: linear vector, not a hash map -- small layer counts, and

129 inline std::vector<std::pair<const void *, ProfileEntry>> &ProfileRegistry()
130 {

135 // Looks up (or creates, on first call) this layer's accumulator. Label
136 // is generated from the layer's own GetInputSize()/GetOutputSize() --
137 // works for any layer type with those methods (DiscriminativePCLayer,
138 // ConvPCLayer, ...) via duck-typed template, no coupling to a specific
139 // class or forward declaration needed.
143 template <typename LayerT>
144 inline ProfileAccumulator &LayerProfile(LayerT *layer)
145 {
146 auto &registry = ProfileRegistry();
147 for (auto &entry : registry)
148 if (entry.first == layer)
149 return entry.second.acc;
150
151 ProfileEntry entry;
152 entry.label = "layer(" + std::to_string(layer->GetInputSize()) + "->"
153 + std::to_string(layer->GetOutputSize()) + ") @ " + std::to_string((size_t)layer);
154 registry.emplace_back(layer, std::move(entry));
155 return registry.back().second.acc;
156 }

158 // Call once, after the training run, with an INDEPENDENTLY measured
159 // wall-clock duration (std::chrono around the whole run in the harness,
160 // NOT derived from these accumulators) -- this is the reconciliation

167 inline void PrintAllProfiles(double wallClockSeconds)
168 {
169 double grandTotal = 0.0;
170 for (auto &entry : ProfileRegistry())
171 {
172 entry.second.acc.Print(entry.second.label);
173 grandTotal += entry.second.acc.PhaseSum();
174 }
175 std::cout << "\n=== Reconciliation ===\n";
176 std::cout << " Wall-clock (measured around whole run): " << std::fixed << std::setprecision(4)
177 << wallClockSeconds << " s\n";
178 std::cout << " Sum of all layers' phase sums: " << grandTotal << " s\n";
179 std::cout << " Unaccounted-for time: " << (wallClockSeconds - grandTotal) << " s"
180 << " (" << (wallClockSeconds > 0 ? 100.0 * (wallClockSeconds - grandTotal) / wallClockSeconds : 0.0)
181 << "% -- data loading, Python/pybind overhead, timer overhead itself, etc.)\n";
182 }
